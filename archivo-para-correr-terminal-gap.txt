#A.6 Chiral 4-polytopes

candS3 := function(s1, s2, G, P)
    local eG, resultado, j;

    eG := Elements(G);
    resultado := [];

    for j in [1..Length(eG)] do
        if Order(s1 * s2 * eG[j]) = 2 and Order(s2 * eG[j]) = 2 and Order(eG[j]) > 2 then
            Add(resultado, Factorization(P, eG[j]));
        fi;
    od;

    return resultado;
end;

StabE:=function(v,e,G)
	local eP, lStabE, i;

	eP:=Elements(G);
	lStabE:=[];
	for i in [1..Length(eP)] do
		if (Length(Cycle(eP[i],v))>1 and Cycle(eP[i],v)[2]=e and Length(Cycle(eP[i],e))>1 and Cycle(eP[i],e)[2]=v) then
			Add(lStabE,eP[i]);
		fi;
		if (Length(Cycle(eP[i],v))=1 and Length(Cycle(eP[i],e))=1) then
			Add(lStabE,eP[i]);
		fi;
	od;
	return lStabE;
end;

face:=function(f,G)
	local eG, fb,  j, f1, f2;
	
	eG := Elements(G);
	fb:=[];
	for j in [1..Length(eG)] do
		f1:=OnTuples( f, eG[j] );
		f2:=Reversed(f1);
		Add(fb,f1);
		Add(fb,f2);
	od;
	return fb;
end;

StabF:=function(f,G,GS1)
	local lStabF, i, eG;

	eG := Elements(G);
	lStabF:= [];
	for i in [1..Length(eG)] do
		if OnTuples( f, eG[i] ) in face(f,GS1) then
			Add(lStabF,eG[i]);
		fi;
	od;
	return lStabF;
end;

celda:=function(f,G,U)
	local transGU, j, cel;
	
	transGU :=Elements(RightTransversal(G,U));
	cel:=[];
	for j in [1..Length(transGU)] do
		Add(cel,OnTuples( f, transGU[j] ));
	od;
	return cel;
end;

nf := function(fa, k)
    local ln, i, n;
    n := Length(fa);
    ln := [];
    for i in [1..n] do
        Add(ln, fa[(i + k - 1) mod n + 1]);
    od;
    return ln;
end;

sf:= function(fa)
    local bc1, k, f1;

    bc1 := [];
    Add(bc1, fa);             # Original
    Add(bc1, Reversed(fa));   # Reverso del original

    for k in [1..(Length(fa)-1)] do
        f1 := nf(fa, k);
        Add(bc1, f1);             # Rotación
        Add(bc1, Reversed(f1));  # Reverso de la rotación
    od;

    return bc1;
end;

StabC:= function(G, Celda, bc)
    local np, eG, lStabC, i, k, j, cont;

    np := Length(Celda);     
    eG := Elements(G);      
    lStabC := [];           

    for i in [1..Length(eG)] do
        cont := 0;
        for k in [1..np] do
            for j in [1..Length(bc)] do
                if OnTuples(Celda[k], eG[i]) in bc[j] then
                    cont := cont + 1;
                    break;  # Ya está en bc, no necesita revisar más
                fi;
            od;
        od;
        if cont = np then
            Add(lStabC, eG[i]);
        fi;
    od;

    return lStabC;
end;

Stabe:=function(v,e,G)
	local eP, lStabE, i;

	eP:=Elements(G);
	lStabE:=[];
	for i in [1..Length(eP)] do
		if v*eP[i]=e and e*eP[i]=v then
			Add(lStabE,eP[i]);
		fi;
		if v*eP[i]=v and e*eP[i]=e  then
			Add(lStabE,eP[i]);
		fi;
	od;
	return lStabE;
end;

Stabf:=function(f,G,GS1)
	local lStabF, i, j, eG, im;

	eG := Elements(G);
	lStabF:= [];
	for i in [1..Length(eG)] do
		im:=[]; #Applying the symmetry eG[i] to f.
		for j in [1..Length(f)] do
			Add(im,f[j]* eG[i]);
		od;
		if im in sf(f) then
			Add(lStabF,eG[i]);
		fi;	
	od;
	return lStabF;
end;

cel:=function(f,G,U)
	local transGU, i, j, cell, im;
	
	transGU :=Elements(RightTransversal(G,U));
	cell:=[];
	for i in [1..Length(transGU)] do
		im:=[]; #Applying the symmetry transGU[i] ) to f.
		for j in [1..Length(f)] do
			Add(im,f[j]* transGU[i]);
		od;
		Add(cell, im);
	od;
	return cell;
end;

im:=function(Celda,S)
	local m, l, ImC, im;
	
	ImC:=[];
	for m in [1..Length(Celda)] do
		im:=[]; #Image of each face.
		for l in [1..Length(Celda[1])] do
			Add(im,Celda[m][l]*S);
		od;
		Add(ImC,im);
	od;
	return ImC;
end;

Stabc:= function(G, Celda, bc)
    local np, eG, lStabC, i, j, k, cont;

    np := Length(Celda);     
    eG := Elements(G);      
    lStabC := [];           

    for i in [1..Length(eG)] do
        cont := 0;
        for k in [1..np] do
            for j in [1..Length(bc)] do
               if im(Celda,eG[i])[k] in bc[j] then
                    cont := cont + 1;
                    break;  # Ya está en bc, no necesita revisar más
                fi;
            od;
        od;
        if cont = np then
            Add(lStabC, eG[i]);
        fi;
    od;

    return lStabC;
end;


#The next function is more efficient than the previous.
Stabc := function(G, Celda, bc)
    local np, eG, lStabC, i, j, k, cont, found;

    np := Length(Celda);     
    eG := Elements(G);      
    lStabC := [];           

    for i in [1..Length(eG)] do
        cont := 0;
        for k in [1..np] do
            found := false;
            for j in [1..Length(bc)] do
                if im(Celda, eG[i])[k] in bc[j] then
                    cont := cont + 1;
                    found := true;
                    break;  
               fi;
            od;
            if not found then #Aquí se puede sustituir not found por found=false.
                break;
            fi;
        od;
        if cont = np then
            Add(lStabC, eG[i]);
        fi;
    od;

    return lStabC;
end;
